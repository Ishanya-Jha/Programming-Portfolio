#include <iostream>
#include <deque> 
using namespace std;

void printKMax(int arr[], int n, int k){
	deque<int> d;

for (int i = 0; i < n; i++) {
    while (!d.empty() && d.front() <= i - k) {
        d.pop_front();
    }

    while (!d.empty() && arr[d.back()] <= arr[i]) {
        d.pop_back();
    }

    d.push_back(i);

    if (i >= k - 1) {
        cout << arr[d.front()] << " ";
    }
}

cout << endl;
}
